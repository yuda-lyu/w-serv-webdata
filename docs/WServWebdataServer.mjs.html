//isGiven, 有給(不為undefined與null), 有給卻無效之設定須拋錯, 不可靜默改用預設
let isGiven = (v) => {
    return v !== undefined && v !== null
}


/**
 * 解析與檢查全部設定, 無副作用(不呼叫使用端函數、不修改輸入), 設定錯誤時拋錯; 資料通道之規則由Exec之parseOpt擁有, 資料同步之規則由Sync之checkOpt擁有
 *
 * @param {Object} [opt={}] 輸入設定物件, 鍵同WServWebdataServer之opt
 * @returns {Object} 回傳正規化後之設定, 供建構各元件使用(不再重讀原始opt)
 */
function parseOpt(opt = {}) {

    //fpTableTags, 有給須為非空字串, 否則會靜默改寫行程工作目錄下之預設檔案
    let fpTableTags = get(opt, 'fpTableTags', null)
    if (isGiven(fpTableTags) && !isestr(fpTableTags)) {
        throw new Error('invalid opt.fpTableTags')
    }
    if (!isestr(fpTableTags)) {
        fpTableTags = './tableTags.json'
    }

    //genTag, 有給須為函數
    let genTag = get(opt, 'genTag', null)
    if (isGiven(genTag) && !isfun(genTag)) {
        throw new Error('invalid opt.genTag')
    }
    if (!isfun(genTag)) {
        genTag = () => {
            return now2str() + '|' + genID(6)
        }
    }

    //useDbOrm, 預設false, 開啟資料通道須由使用端明確指定, 開啟時operOrm、tableNamesExec、authorizeExec為必填
    let useDbOrm = get(opt, 'useDbOrm', null)
    let useDbOrmGiven = isbol(useDbOrm)
    if (!useDbOrmGiven) {
        useDbOrm = false
    }

    //kpOrm, 輸入外部ORM實體
    let kpOrm = get(opt, 'kpOrm', null)

    //operOrm, ORM的泛用接口procCommon
    let operOrm = get(opt, 'operOrm', null)

    //tableNamesExec, 指定能被呼叫的表名, 各表名需隸屬於ORM, 才能被ORM的泛用接口procCommon處理
    let tableNamesExec = get(opt, 'tableNamesExec', null)

    //methodsExec
    let methodsExec = get(opt, 'methodsExec', null)

    //authorizeExec, 表操作之授權函數, 預設拒絕, 回傳嚴格等於true才放行
    let authorizeExec = get(opt, 'authorizeExec', null)

    //tableNamesSync, 指定需同步表名, 機敏資訊表記得過濾, 各表名需隸屬於ORM, 才能監聽其change事件
    let tableNamesSync = get(opt, 'tableNamesSync', null)

    //getUserIdByToken, 由外部提供呼叫函數, 將提供各函數的輸入用以取得使用userID
    let getUserIdByToken = get(opt, 'getUserIdByToken', null)

    //kpFunExt, 容器與值之檢查由Exec之parseOpt擁有
    let kpFunExt = get(opt, 'kpFunExt', null)

    //check, 未指定useDbOrm(非布林)卻給了資料通道或資料同步之設定時拋錯, 否則該些設定會因useDbOrm預設false而靜默不生效
    if (!useDbOrmGiven) {
        let kpOptDbOrm = { operOrm, tableNamesExec, methodsExec, authorizeExec, kpOrm, tableNamesSync }
        each(kpOptDbOrm, (v, k) => {
            if (v !== null) {
                throw new Error(`invalid opt.useDbOrm when opt.${k} is given`)
            }
        })
    }

    //check, useDbOrm=true時kpOrm須為物件
    if (useDbOrm && !isobj(kpOrm)) {
        throw new Error('invalid opt.kpOrm when useDbOrm=true')
    }

    //check, useDbOrm=true時tableNamesSync有給須為陣列且各元素為非空字串, 與tableNamesExec對稱
    if (useDbOrm && isGiven(tableNamesSync)) {
        if (!isarr(tableNamesSync)) {
            throw new Error('invalid opt.tableNamesSync when useDbOrm=true')
        }
        each(tableNamesSync, (v, i) => {
            if (!isestr(v)) {
                throw new Error(`invalid opt.tableNamesSync[${i}] when useDbOrm=true`)
            }
        })
        tableNamesSync = [...tableNamesSync] //快照, 避免建構期間外部修改
    }

    //parseExecOpt, 資料通道之設定
    let oe = parseExecOpt({
        getUserIdByToken,
        useDbOrm,
        operOrm,
        tableNames: tableNamesExec,
        methods: methodsExec,
        authorizeExec,
        kpFunExt,
    })

    //checkSyncOpt, 資料同步之設定, 回傳是否同步(kpOrm為空物件時不同步)
    let syncing = false
    if (useDbOrm) {
        syncing = checkSyncOpt(tableNamesSync, kpOrm)
    }

    //check, 同步時各同步表須有經授權之表操作select, 前端同步取資料即呼叫此表操作; kpFunExt提供之select不算(不經authorizeExec)
    if (syncing) {
        each(tableNamesSync, (tableName) => {
            if (!oe.tableNames.includes(tableName) || !oe.methods.includes('select')) {
                throw new Error(`invalid opt.tableNamesSync when table operation[${tableName}.select] does not exist`)
            }
        })
    }

    return {
        fpTableTags,
        genTag,
        useDbOrm,
        kpOrm,
        tableNamesSync,
        optExec: {
            getUserIdByToken: oe.getUserIdByToken,
            useDbOrm: oe.useDbOrm,
            operOrm: oe.operOrm,
            tableNames: oe.tableNames,
            methods: oe.methods,
            authorizeExec: oe.authorizeExec,
            kpFunExt: oe.kpFunExt,
        },
    }
}


/**
 * 伺服器端之資料控制與同步器
 *
 * 資料通道(useDbOrm=true時開啟)開放給前端呼叫之函數有三類：表操作(由tableNamesExec與methodsExec組成，例如'tabA.select')、kpFunExt、系統函數。前端每次呼叫依序經過getUserIdByToken(取得userId)、authorizeExec(僅表操作)，再執行operOrm或kpFunExt之函數。授權與拒絕之規則如下：
 *
 * 1. 資料通道是資料表直接開放給前端之處，表操作之授權在此，由authorizeExec負責。採預設拒絕，回傳值須嚴格等於true才放行，回傳其他值(例如false、1、'true'、undefined)一律回應'permission denied'且不呼叫operOrm。operOrm可能亦被伺服端程式呼叫而分不出呼叫來源，故不負責授權。kpFunExt之函數不經authorizeExec，須各自把關。
 * 2. 各關卡(getUserIdByToken、authorizeExec、operOrm、kpFunExt之函數)拋錯或reject時，其值原樣成為前端收到之拒絕值，故欲回傳語系鍵須reject字串，Error物件經序列化後前端收到'Error: 訊息'字串。各關卡之回傳值與拒絕值須為可序列化之格式，此為設計要求，不支援之格式(例如循環參照、BigInt)本就不在設計範圍內，前端必然失敗。
 * 3. 僅getUserIdByToken之拒絕會發出error事件(其值可能為undefined或null)；authorizeExec、operOrm與kpFunExt之拒絕不發出error事件，需於伺服端紀錄者請於函數內自行處理。
 * 4. authorizeExec只能放行或拒絕整次呼叫，無法逐筆過濾資料或限制欄位。同步資料時前端呼叫該表之表操作select且不帶參數，故tableNamesSync內之表須在tableNamesExec中、methodsExec須含'select'(kpFunExt提供之同名函數不算)，否則建構時拋錯；其select須對所有連線者放行，否則被拒絕之前端取不到該表資料，且每次重抓皆發出error事件。需逐筆控管可見範圍者請改用kpFunExt提供且不列入同步。
 * 5. authorizeExec收到之input與傳給operOrm者為同一物件，僅供判斷，不支援修改。授權與operOrm之執行非原子操作。methodsExec含'mix'時，允許mix等同允許該表之新增、修改與刪除。
 * 6. getUserIdByToken須完整驗證token(不可只解碼)，其token取自請求內容，與通訊層(例如w-converhp之verifyConn)所驗證者為不同來源。getUserIdByToken或authorizeExec不結束(promise不settle)時，該次請求將懸置。
 * 7. 函數名須能被前端綁定(前端依函數名之路徑綁定，解析同lodash之toPath，例如'tabA[select]'即['tabA','select'])：路徑不得為空、不得有空段、各段不得為'__proto__'、'constructor'、'prototype'；kpFunExt不得以'[sys:'開頭；kpFunExt之頂層名不得為表名(例如表名為'tabA'時，'tabA'、'tabA.count'、'tabA[select]'皆不可)；不同函數名(含表操作與kpFunExt)不得解析為同一路徑(例如'a.b'與'a["b"]')，亦不得互為前綴(例如'a'與'a.b')。違反者於建構時拋錯。
 * 8. 設定錯誤於建構任何元件之前拋錯，此時尚未擴充傳入之事件物件、未註冊任何監聽、未寫入時間戳檔、未呼叫使用端之genTag。設定錯誤包含：未指定useDbOrm(非布林值)卻給了operOrm、tableNamesExec、methodsExec、authorizeExec、kpOrm或tableNamesSync；useDbOrm=true而operOrm、authorizeExec非函數，tableNamesExec非陣列，kpOrm非物件，methodsExec、tableNamesSync有給但非陣列，或tableNamesExec、methodsExec、tableNamesSync之元素非非空字串；同步之表不在kpOrm中或無法監聽change事件，或無表操作select；getUserIdByToken、genTag有給但非函數，fpTableTags有給但非非空字串，kpFunExt有給但非純物件或其值非函數；以及違反第7點之函數名。執行期異常(例如使用端genTag拋錯或回傳非字串、kpOrm之on拋錯)會先停止廣播之計時器、釋放同步元件並移除已掛於kpOrm之監聽，但傳入之事件物件已被擴充，不可再以同一事件物件重新建構。
 * 9. 停止服務時呼叫clearBroadcast(停止廣播)與clearSync(釋放資料同步)。clearSync會移除本套件掛於kpOrm之change監聽、取消待推播之時間戳並將尚未落檔之時間戳寫入檔案，之後kpOrm之變更不再更新時間戳與推播，可重複呼叫。釋放後傳入之事件物件上仍有廣播與資料通道之監聽，不可再以同一事件物件重新建構。
 *
 * @class
 * @param {Object} instWConverServer 輸入通訊服務實體物件，可使用例如WConverhpServer等建立
 * @param {Object} [opt={}] 輸入設定物件，預設{}
 * @param {Boolean} [opt.useDbOrm=false] 輸入是否使用資料庫ORM技術，給予true代表開啟資料通道(前端可直接呼叫由tableNamesExec與methodsExec組成之表操作，例如'tabA.select')與自動同步資料庫至前端功能，此時operOrm、tableNamesExec、authorizeExec為必填，未給或無效時拋錯。未指定(非布林值)卻給了operOrm、tableNamesExec、methodsExec、authorizeExec、kpOrm或tableNamesSync時拋錯，預設false
 * @param {Object} [opt.kpOrm=null] 輸入各資料表的操作物件，用以提供由tableNamesSync指定資料表的change事件，使能監聽與觸發資料變更事件，key為表名而值為該表的操作器實體，操作器實體可使用例如WOrmMongodb等建立。useDbOrm為true時必填且須為物件，否則拋錯；為空物件時不同步，預設null
 * @param {Function} [opt.operOrm=null] 輸入各資料表的操作通用接口函數，用以提供操作由tableNamesExec指定資料表的例如'select'、'insert'、'save'、'del'函數。加上由kpFunExt提供的函數，就為全部可由前端執行的函數。前端呼叫表操作時須先通過authorizeExec授權才會呼叫operOrm，operOrm不負責授權。useDbOrm為true時必填，預設null
 * @param {Array} [opt.tableNamesExec=null] 輸入指定能被操作的表名陣列，各元素須為非空字串。useDbOrm為true時必填(可為空陣列)，預設null
 * @param {Array} [opt.tableNamesSync=null] 輸入指定能被同步的表名陣列，各元素須為非空字串，各表須在kpOrm中且可監聽change事件，並須有表操作select(見上方第4點)，未給時不同步，預設null
 * @param {Array} [opt.methodsExec=['select','insert','save','del']] 輸入指定綁定操作器的方式陣列，各元素須為非空字串，可選'select'、'insert'、'save'、'del'、'delAll'。未給或null時取預設['select', 'insert', 'save', 'del']；useDbOrm為true時若有給須為陣列，否則拋錯
 * @param {Function} [opt.authorizeExec=null] 輸入資料表操作之授權函數，useDbOrm為true時必填，未給或非函數時拋錯。前端呼叫表操作時，會於呼叫operOrm之前傳入物件{ userId, tableName, methodName, input }，可使用sync或async函數，回傳值須嚴格等於true才放行。userId為getUserIdByToken所得，取不到時為空字串，宜拒絕；methodName為methodsExec所列之方法，宜以白名單判斷，其中'mix'之input可含多筆操作須自行解析，規則詳見上方說明，預設null
 * @param {Function} [opt.getUserIdByToken=async()=>''] 輸入取得使用者ID的回調函數，傳入參數為前端提供之token字串，可使用sync或async函數，回傳非字串時視為空字串，拋錯或reject時拒絕該次呼叫，規則詳見上方說明；有給但非函數時拋錯，預設async()=>''
 * @param {Object} [opt.kpFunExt=null] 輸入額外擴充執行函數物件，須為純物件，key為函數名而值為函數(值非函數時拋錯)，函數第1參數為userId(取不到時為空字串)，之後為前端給予之參數。kpFunExt之函數不經authorizeExec須各自把關，命名限制詳見上方說明，預設null
 * @param {String} [opt.fpTableTags='./tableTags.json'] 輸入儲存各資料表時間戳檔案路徑串，有給但非非空字串時拋錯，預設'./tableTags.json'
 * @param {Function} [opt.genTag=()=>'{random string}'] 輸入產生不重複識別碼函數，須回傳非空字串(回傳其他值時拋錯)，有給但非函數時拋錯，預設()=>'{random string}'
 * @returns {Object} 回傳事件物件，可監聽error事件，可使用函數clearBroadcast、clearSync停止服務(見上方第9點)
 * @example
 *

 * let tableNamesExec = ['tabA', 'tabB']

 *         name: 'rosemary',

 *         id: 'id-tabA-kettle',

 *         id: 'id-tabB-rosemary',
 *         name: 'rosemary',

 *     n++

 *         uploadFile: async (userId, { name, u8a }) => {

 *             return r

function WServWebdataServer(instWConverServer, opt = {}) {

    //check
    if (!iseobj(instWConverServer)) {
        console.log('instWConverServer is not an effective object, and set instWConverServer to an EventEmitter')
        instWConverServer = evem()
    }
    if (!haskey(instWConverServer, 'emit') || !isfun(instWConverServer.on)) {
        throw new Error(`instWConverServer is not an EventEmitter`)
    }

    //parseOpt, 解析與檢查全部設定, 設定錯誤於此拋出: 此時尚未擴充傳入之事件物件、未註冊任何監聽、未寫檔、未呼叫使用端之genTag
    let { fpTableTags, genTag, useDbOrm, kpOrm, tableNamesSync, optExec } = parseOpt(opt)

    //擴充同步資料功能
    instWConverServer = new WSyncWebdataServer(instWConverServer, {
        fpTableTags,
        genTag, //供updateTableTag使用
    })

    //擴充廣播功能, execute監聽器之註冊順序須為廣播先於資料通道, 否則[sys:polling]會被資料通道以invalid token搶先回應
    instWConverServer = new WServBroadcastServer(instWConverServer)

    //kpFunSys, 提供同步資料[sys:getTableTags]函數, 與使用端之kpFunExt分開傳入, 使kpFunExt之命名檢查不涉及系統函數
    let kpFunSys = {
        '[sys:getTableTags]': () => {
            let nowTableTags = instWConverServer.getTableTags()
            // console.log('nowTableTags', nowTableTags)
            return {
                mode: 'syncKpTable',
                data: pickTables(tableNamesSync, nowTableTags),
            }
        },
    }

    //設定錯誤已於parseOpt拋出, 此處只剩執行期異常(例如使用端genTag拋錯、kpOrm之on拋錯), 拋錯前先釋放已建構之元件: 停止廣播之計時器(否則呼叫端攔下錯誤後程序仍被計時器撐住而無法結束), 釋放同步元件(殘留之實體不再更新時間戳或寫檔; 已掛於kpOrm之監聽由Sync回滾或經其包裝之clearSync移除)
    try {

        //WServWebdataServerExec, 以parseOpt之結果建構, 不再重讀原始opt
        instWConverServer = new WServWebdataServerExec(
            instWConverServer,
            {
                ...optExec,
                kpFunSys,
            },
        )

        //useDbOrm
        if (useDbOrm) {

            //WServWebdataServerSync
            instWConverServer = new WServWebdataServerSync(
                instWConverServer,
                tableNamesSync,
                kpOrm,
                {
                    genTag, //供初始化tableTagsSrv用
                },
            )

        }

    }
    catch (err) {
        instWConverServer.clearBroadcast()
        instWConverServer.clearSync()
        throw err
    }

    return instWConverServer
}


export default WServWebdataServer
